var GEMINI_KEY='COLLE_TA_CLE_GEMINI_ICI'; var DATA_SHEET='_Data'; var COACHING_SHEET='🗣 Coaching & Briefings'; var PIP_SHEET='PIPs Conducted'; var PLANNING_SHEET='Raw Data'; var TZ=Session.getScriptTimeZone() || 'Africa/Casablanca'; function doGet(e){ var cb=(e.parameter.callback||'cb').replace(/[^a-zA-Z0-9_]/g,''); var act=e.parameter.action||'data'; if(act==='ai')return handleAI(e,cb); return handleData(e,cb); } function handleData(e,cb){ try{ var ss=SpreadsheetApp.getActiveSpreadsheet(); var dataSheet=ss.getSheetByName(DATA_SHEET); if(!dataSheet)return out(cb,{status:'error',error:'Sheet '+DATA_SHEET+' introuvable'}); var values=dataSheet.getDataRange().getValues(); var hdrRow=findHeaderRow(values,['Week #','Agent Name']); if(hdrRow<0)return out(cb,{status:'error',error:'Headers Week # / Agent Name introuvables dans '+DATA_SHEET}); var hdr=values[hdrRow].map(cleanHeader); var rows=[]; var weekMap={}; var agentMap={}; for(var r=hdrRow+1;r<values.length;r++){ var raw={};hdr.forEach(function(h,i){raw[h]=values[r][i];}); if(isBlank(raw['Week #'])||isBlank(raw['Agent Name']))continue; var norm=normalizeDataRow(raw); rows.push(norm); if(norm.Semaine){weekMap[norm.Semaine]={num:norm.Semaine,month:norm.Month||'',start:norm['Week Start']||'',end:norm['Week End']||''};} if(norm.Agent){agentMap[norm.Agent.toLowerCase()]={name:norm.Agent,role:norm['Line of Business']||'Customer Support',type:guessType(norm['Line of Business'])};} } var weeks=Object.keys(weekMap).map(function(k){return weekMap[k];}).sort(function(a,b){return a.num-b.num;}); var agents=Object.keys(agentMap).map(function(k){return agentMap[k];}).sort(function(a,b){return a.name.localeCompare(b.name);}); return out(cb,{status:'ok',data:rows,coaching:readCoaching(ss),pips:readPIPs(ss),planning:readPlanning(ss),config:{weeks:weeks,agents:agents},updatedAt:new Date().toISOString()}); }catch(err){return out(cb,{status:'error',error:err.toString()});} } function normalizeDataRow(raw){ var week=toNumber(raw['Week #']); var wkStart=fmtDate(raw['Week Start']); var wkEnd=fmtDate(raw['Week End']); var qGlobal=percent(raw['Quality score (Global)']); var qEmail=percent(raw['Quality score Email']); var qCalls=percent(raw['Quality score Calls']); return { 'Agent': String(raw['Agent Name']||'').trim(), 'Semaine': week, 'Month': monthName(raw['Week Start']), 'Line of Business': String(raw['LOB']||'').trim(), 'Team Leader': String(raw['Team Leader']||'').trim(), 'Week Start': wkStart, 'Week End': wkEnd, 'Emails Handled': safeNumber(raw['Emails']), 'Emails': safeNumber(raw['Emails']), 'Calls': safeNumber(raw['Inbound calls']), 'Inbound calls': safeNumber(raw['Inbound calls']), 'Outbound Calls': safeNumber(raw['Outbound Calls']), 'Chats': safeNumber(raw['Chat']), 'Chat': safeNumber(raw['Chat']), 'Jira Tickets': safeNumber(raw['Jira Tickets created']), 'Jira Tickets created': safeNumber(raw['Jira Tickets created']), 'Total Contacts (auto)': safeNumber(raw['Total Contacts (auto)']), 'Productivity': round2(raw['Productivity']), 'AHT (min)': ahtMinutes(raw['Avg AHT (min )']), 'Avg AHT (min)': ahtMinutes(raw['Avg AHT (min )']), 'Quality score (Global)': qGlobal, 'Qualité Email (%)': qEmail, 'Qualite (%)': qEmail, 'Quality score Email': qEmail, 'Qualité Téléphone (%)': qCalls, 'Quality score Calls': qCalls, 'Total hrs worked': safeNumber(raw['Total hrs worked']) }; } function readCoaching(ss){ var sh=ss.getSheetByName(COACHING_SHEET);if(!sh)return []; var v=sh.getDataRange().getValues(); var outRows=[]; for(var i=3;i<v.length;i++){ var agent=String(v[i][8]||'').trim(); if(!agent)continue; outRows.push({agent:agent,date:fmtDate(v[i][9]),coach:String(v[i][10]||'').trim(),focus:String(v[i][11]||'').trim(),issue:String(v[i][12]||'').trim(),actionPlan:String(v[i][13]||'').trim(),followUpDate:fmtDate(v[i][14]),status:String(v[i][15]||'').trim()}); } return outRows; } function readPIPs(ss){ var sh=ss.getSheetByName(PIP_SHEET);if(!sh)return []; var v=sh.getDataRange().getValues(); var hdrRow=findHeaderRow(v,['Agent']);if(hdrRow<0)return []; var hdr=v[hdrRow].map(cleanHeader), rows=[]; for(var r=hdrRow+1;r<v.length;r++){ var raw={};hdr.forEach(function(h,i){raw[h]=v[r][i];}); if(isBlank(raw['Agent']))continue; rows.push({agent:String(raw['Agent']||'').trim(),start:fmtDate(raw['Start Date']||raw['PIP Start']||raw['Date']),due:fmtDate(raw['Due Date']||raw['Follow-Up Date']||raw['End Date']),status:String(raw['Status']||'En cours').trim(),objective:String(raw['Objective']||raw['Reason']||raw['KPI Focus']||'').trim(),closureDate:fmtDate(raw['Closure Date']||raw['Closed Date']),closureNote:String(raw['Closure Note']||raw['Closure']||'').trim()}); } return rows; } function readPlanning(ss){ var sh=ss.getSheetByName(PLANNING_SHEET);if(!sh)return []; var v=sh.getDataRange().getValues(); var hdrRow=findHeaderRow(v,['Agent','Day','Shift']);if(hdrRow<0)return []; var hdr=v[hdrRow].map(cleanHeader),rows=[]; for(var r=hdrRow+1;r<v.length;r++){ var raw={};hdr.forEach(function(h,i){raw[h]=v[r][i];}); var agent=String(raw['Agent']||'').trim(); if(!agent||agent===' ')continue; rows.push({week:String(raw['Week']||'').trim(),agent:agent,dept:String(raw['Department']||'').trim(),day:String(raw['Day']||'').trim(),shift:String(raw['Shift']||'OFF').trim()}); } return rows; } function handleAI(e,cb){ var prompt=e.parameter.prompt||''; if(!prompt)return out(cb,{error:'Prompt vide'}); try{ var url='https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key='+GEMINI_KEY; var res=UrlFetchApp.fetch(url,{method:'post',contentType:'application/json',payload:JSON.stringify({contents:[{role:'user',parts:[{text:prompt}]}],generationConfig:{maxOutputTokens:1200,temperature:0.4}}),muteHttpExceptions:true}); var j=JSON.parse(res.getContentText()); var txt=j.candidates&&j.candidates[0]&&j.candidates[0].content&&j.candidates[0].content.parts&&j.candidates[0].content.parts[0]?j.candidates[0].content.parts[0].text:''; return out(cb,{text:txt||'Vide',status:'ok'}); }catch(err){return out(cb,{error:err.toString()});} } function findHeaderRow(values, required){ for(var r=0;r<Math.min(values.length,20);r++){ var row=values[r].map(cleanHeader); var ok=required.every(function(h){return row.indexOf(h)>=0;}); if(ok)return r; } return -1; } function cleanHeader(x){return String(x||'').replace(/\s+/g,' ').trim();} function isBlank(v){return v===null||v===''||String(v).trim()==='';} function toNumber(v){var n=Number(v);return isNaN(n)?'':n;} function safeNumber(v){var n=Number(v);return isNaN(n)?'':n;} function round2(v){var n=Number(v);return isNaN(n)?'':Math.round(n*100)/100;} function percent(v){if(isBlank(v)||String(v).indexOf('-%')>=0||String(v)==='-')return '';var n=Number(String(v).replace('%','').replace(',','.'));if(isNaN(n))return '';if(n<=1)n=n*100;return Math.round(n*100)/100;} function ahtMinutes(v){if(isBlank(v)||String(v)==='-')return '';if(Object.prototype.toString.call(v)==='[object Date]')return Math.round((v.getHours()*60+v.getMinutes()+v.getSeconds()/60)*100)/100;var s=String(v).trim();if(s.indexOf(':')>0){var p=s.split(':').map(Number);return Math.round(((p[0]||0)*60+(p[1]||0)+(p[2]||0)/60)*100)/100;}var n=Number(s.replace(',','.'));if(isNaN(n))return '';if(n>0&&n<1)n=n*1440;return Math.round(n*100)/100;} function fmtDate(v){if(!v)return '';if(Object.prototype.toString.call(v)==='[object Date]')return Utilities.formatDate(v,TZ,'yyyy-MM-dd');if(typeof v==='number')return Utilities.formatDate(new Date(Math.round((v-25569)*86400*1000)),TZ,'yyyy-MM-dd');return String(v);} function monthName(v){var d=null;if(Object.prototype.toString.call(v)==='[object Date]')d=v;else if(typeof v==='number')d=new Date(Math.round((v-25569)*86400*1000));if(!d)return '';return ['Janvier','Février','Mars','Avril','Mai','Juin','Juillet','Août','Septembre','Octobre','Novembre','Décembre'][d.getMonth()];} function guessType(lob){lob=String(lob||'').toLowerCase();return lob.indexOf('sm')>=0||lob.indexOf('social')>=0?'sm':'ep';} function out(cb,obj){return ContentService.createTextOutput(cb+'('+JSON.stringify(obj)+');').setMimeType(ContentService.MimeType.JAVASCRIPT);}
